### 阶段 1：解耦核心系统（2-3周）

#### 1.1 引入事件总线系统
**目标**: 减少组件间直接依赖

```cpp
// EventBus.h - 新建文件
UCLASS()
class AURA_API UEventBus : public UGameInstanceSubsystem
{
    GENERATED_BODY()
public:
    // 订阅事件
    template<typename T>
    FDelegateHandle Subscribe(TFunction<void(const T&)> Handler)
    {
        if (!EventHandlers.Contains(T::StaticStruct()))
        {
            EventHandlers.Add(T::StaticStruct(), TArray<TFunction<void(const void*)>>());
        }
        
        auto& Handlers = EventHandlers[T::StaticStruct()];
        Handlers.Add([Handler](const void* Data) {
            Handler(*static_cast<const T*>(Data));
        });
        
        return FDelegateHandle();  // 简化版
    }
    
    // 发布事件
    template<typename T>
    void Publish(const T& Event)
    {
        if (EventHandlers.Contains(T::StaticStruct()))
        {
            for (auto& Handler : EventHandlers[T::StaticStruct()])
            {
                Handler(&Event);
            }
        }
    }
    
private:
    TMap<UScriptStruct*, TArray<TFunction<void(const void*)>>> EventHandlers;
};

// 事件定义
USTRUCT(BlueprintType)
struct FAttributeChangedEvent
{
    GENERATED_BODY()
    
    UPROPERTY()
    FGameplayTag AttributeTag;
    
    UPROPERTY()
    float OldValue;
    
    UPROPERTY()
    float NewValue;
};

USTRUCT(BlueprintType)
struct FAbilityActivatedEvent
{
    GENERATED_BODY()
    
    UPROPERTY()
    FGameplayTag AbilityTag;
    
    UPROPERTY()
    AActor* Instigator;
};
```

**使用示例**:
```cpp
// AttributeSet 中发布事件
void UAuraAttributeSet::PostAttributeChange(const FGameplayAttribute& Attribute, float OldValue, float NewValue)
{
    Super::PostAttributeChange(Attribute, OldValue, NewValue);
    
    UEventBus* EventBus = GetWorld()->GetGameInstance()->GetSubsystem<UEventBus>();
    if (EventBus)
    {
        FAttributeChangedEvent Event;
        Event.AttributeTag = /* 获取对应标签 */;
        Event.OldValue = OldValue;
        Event.NewValue = NewValue;
        
        EventBus->Publish(Event);
    }
}

// WidgetController 中订阅事件
void UOverlayWidgetController::BindCallbacksToDependencies()
{
    UEventBus* EventBus = GetWorld()->GetGameInstance()->GetSubsystem<UEventBus>();
    if (EventBus)
    {
        EventBus->Subscribe<FAttributeChangedEvent>([this](const FAttributeChangedEvent& Event) {
            if (Event.AttributeTag == FAuraGameplayTags::Get().Attributes_Vital_Health)
            {
                OnHealthChanged.Broadcast(Event.NewValue);
            }
        });
    }
}
```

**收益**:
- ✅ 减少直接依赖
- ✅ 更容易添加新的监听者
- ✅ 更好的可测试性

#### 1.2 服务定位器模式
**目标**: 统一管理全局服务

```cpp
// ServiceLocator.h - 新建文件
UCLASS()
class AURA_API UServiceLocator : public UGameInstanceSubsystem
{
    GENERATED_BODY()
public:
    // 注册服务
    template<typename T>
    void RegisterService(T* Service)
    {
        Services.Add(T::StaticClass(), Service);
    }
    
    // 获取服务
    template<typename T>
    T* GetService()
    {
        if (UObject** Found = Services.Find(T::StaticClass()))
        {
            return Cast<T>(*Found);
        }
        return nullptr;
    }
    
private:
    UPROPERTY()
    TMap<UClass*, UObject*> Services;
};

// 使用示例
void AAuraGameInstance::Init()
{
    Super::Init();
    
    UServiceLocator* ServiceLocator = GetSubsystem<UServiceLocator>();
    
    // 注册服务
    ServiceLocator->RegisterService(AssetManager);
    ServiceLocator->RegisterService(EventBus);
}

// 获取服务
UAuraAssetManager* AssetManager = GetGameInstance()->GetSubsystem<UServiceLocator>()->GetService<UAuraAssetManager>();
```

### 阶段 2：性能优化（2-3周）

#### 2.1 对象池系统
**目标**: 减少投射物、特效的创建销毁开销

```cpp
// ObjectPool.h - 新建文件
template<typename T>
class TObjectPool
{
public:
    TObjectPool(UWorld* InWorld, TSubclassOf<T> InClass, int32 InitialSize = 10)
        : World(InWorld), ObjectClass(InClass)
    {
        for (int32 i = 0; i < InitialSize; ++i)
        {
            T* NewObject = World->SpawnActor<T>(ObjectClass);
            NewObject->SetActorHiddenInGame(true);
            NewObject->SetActorEnableCollision(false);
            Pool.Add(NewObject);
        }
    }
    
    T* Acquire()
    {
        if (Pool.Num() > 0)
        {
            T* Object = Pool.Pop();
            Object->SetActorHiddenInGame(false);
            Object->SetActorEnableCollision(true);
            return Object;
        }
        
        // 池为空，创建新对象
        return World->SpawnActor<T>(ObjectClass);
    }
    
    void Release(T* Object)
    {
        Object->SetActorHiddenInGame(true);
        Object->SetActorEnableCollision(false);
        Pool.Add(Object);
    }
    
private:
    UWorld* World;
    TSubclassOf<T> ObjectClass;
    TArray<T*> Pool;
};

// ProjectilePoolManager.h
UCLASS()
class AURA_API UProjectilePoolManager : public UGameInstanceSubsystem
{
    GENERATED_BODY()
public:
    AAuraProjectile* AcquireProjectile(TSubclassOf<AAuraProjectile> ProjectileClass);
    void ReleaseProjectile(AAuraProjectile* Projectile);
    
private:
    TMap<TSubclassOf<AAuraProjectile>, TSharedPtr<TObjectPool<AAuraProjectile>>> Pools;
};

// 使用示例
void UAuraProjectileSpell::SpawnProjectile(const FVector& ProjectileTargetLocation, const FGameplayTag& SocketTag)
{
    UProjectilePoolManager* PoolManager = GetWorld()->GetGameInstance()->GetSubsystem<UProjectilePoolManager>();
    AAuraProjectile* Projectile = PoolManager->AcquireProjectile(ProjectileClass);
    
    // 设置投射物
    Projectile->SetActorLocation(SocketLocation);
    Projectile->SetActorRotation(Rotation);
    Projectile->DamageEffectParams = MakeDamageEffectParamsFromClassDefaults();
    
    // 投射物销毁时归还到池
    Projectile->OnDestroyed.AddDynamic(this, &UAuraProjectileSpell::OnProjectileDestroyed);
}

void UAuraProjectileSpell::OnProjectileDestroyed(AActor* DestroyedActor)
{
    UProjectilePoolManager* PoolManager = GetWorld()->GetGameInstance()->GetSubsystem<UProjectilePoolManager>();
    PoolManager->ReleaseProjectile(Cast<AAuraProjectile>(DestroyedActor));
}
```

**收益**:
- ✅ 减少 GC 压力
- ✅ 提升性能 30-50%
- ✅ 减少卡顿

#### 2.2 异步资源加载
**目标**: 避免同步加载造成的卡顿

```cpp
// AsyncAssetLoader.h - 新建文件
UCLASS()
class AURA_API UAsyncAssetLoader : public UGameInstanceSubsystem
{
    GENERATED_BODY()
public:
    // 异步加载资产
    template<typename T>
    void LoadAssetAsync(const FSoftObjectPath& AssetPath, TFunction<void(T*)> OnLoaded)
    {
        if (AssetPath.IsValid())
        {
            TSharedPtr<FStreamableHandle> Handle = StreamableManager.RequestAsyncLoad(
                AssetPath,
                [OnLoaded, AssetPath]()
                {
                    T* LoadedAsset = Cast<T>(AssetPath.ResolveObject());
                    OnLoaded(LoadedAsset);
                }
            );
        }
    }
    
    // 批量异步加载
    void LoadAssetsAsync(const TArray<FSoftObjectPath>& AssetPaths, TFunction<void()> OnAllLoaded)
    {
        TSharedPtr<FStreamableHandle> Handle = StreamableManager.RequestAsyncLoad(
            AssetPaths,
            OnAllLoaded
        );
    }
    
private:
    FStreamableManager StreamableManager;
};

// 使用示例
void AAuraCharacter::LoadAbilityAssets()
{
    UAsyncAssetLoader* AssetLoader = GetGameInstance()->GetSubsystem<UAsyncAssetLoader>();
    
    TArray<FSoftObjectPath> AbilityPaths;
    for (const auto& AbilityClass : StartupAbilities)
    {
        AbilityPaths.Add(FSoftObjectPath(AbilityClass.ToString()));
    }
    
    AssetLoader->LoadAssetsAsync(AbilityPaths, [this]()
    {
        // 所有技能资产加载完成
        AddCharacterAbilities();
    });
}
```
