## 🔧 核心重构方案

### 方案1: 事件总线系统（解耦核心）

#### 问题分析
当前系统使用大量委托进行组件间通信，导致：
- 组件间紧耦合
- 难以追踪事件流
- 添加新监听者需要修改多处代码

#### 解决方案：中央事件总线

```cpp
// EventBus.h - 新建文件
#pragma once

#include "CoreMinimal.h"
#include "Subsystems/GameInstanceSubsystem.h"
#include "GameplayTagContainer.h"
#include "EventBus.generated.h"

/**
 * 事件基类 - 所有事件必须继承此类
 */
USTRUCT(BlueprintType)
struct FAuraEvent
{
    GENERATED_BODY()
    
    UPROPERTY(BlueprintReadOnly)
    FGameplayTag EventTag;
    
    UPROPERTY(BlueprintReadOnly)
    float Timestamp;
    
    FAuraEvent()
        : Timestamp(0.f)
    {}
    
    virtual ~FAuraEvent() = default;
};

/**
 * 属性变化事件
 */
USTRUCT(BlueprintType)
struct FAttributeChangedEvent : public FAuraEvent
{
    GENERATED_BODY()
    
    UPROPERTY(BlueprintReadOnly)
    FGameplayTag AttributeTag;
    
    UPROPERTY(BlueprintReadOnly)
    float OldValue;
    
    UPROPERTY(BlueprintReadOnly)
    float NewValue;
    
    UPROPERTY(BlueprintReadOnly)
    AActor* Instigator;
};

/**
 * 技能激活事件
 */
USTRUCT(BlueprintType)
struct FAbilityActivatedEvent : public FAuraEvent
{
    GENERATED_BODY()
    
    UPROPERTY(BlueprintReadOnly)
    FGameplayTag AbilityTag;
    
    UPROPERTY(BlueprintReadOnly)
    AActor* Instigator;
    
    UPROPERTY(BlueprintReadOnly)
    AActor* Target;
};

/**
 * 伤害事件
 */
USTRUCT(BlueprintType)
struct FDamageEvent : public FAuraEvent
{
    GENERATED_BODY()
    
    UPROPERTY(BlueprintReadOnly)
    float Damage;
    
    UPROPERTY(BlueprintReadOnly)
    bool bCriticalHit;
    
    UPROPERTY(BlueprintReadOnly)
    bool bBlockedHit;
    
    UPROPERTY(BlueprintReadOnly)
    AActor* Attacker;
    
    UPROPERTY(BlueprintReadOnly)
    AActor* Victim;
};

/**
 * 事件总线 - 中央事件分发系统
 */
UCLASS()
class AURA_API UEventBus : public UGameInstanceSubsystem
{
    GENERATED_BODY()
    
public:
    // 订阅事件（C++）
    template<typename TEvent>
    FDelegateHandle Subscribe(TFunction<void(const TEvent&)> Handler)
    {
        static_assert(TIsDerivedFrom<TEvent, FAuraEvent>::Value, "TEvent must derive from FAuraEvent");
        
        const FName EventName = TEvent::StaticStruct()->GetFName();
        
        if (!EventHandlers.Contains(EventName))
        {
            EventHandlers.Add(EventName, TArray<TFunction<void(const void*)>>());
        }
        
        auto& Handlers = EventHandlers[EventName];
        Handlers.Add([Handler](const void* Data) {
            Handler(*static_cast<const TEvent*>(Data));
        });
        
        // 返回句柄用于取消订阅
        FDelegateHandle Handle;
        Handle.Reset();
        return Handle;
    }
    
    // 发布事件（C++）
    template<typename TEvent>
    void Publish(const TEvent& Event)
    {
        static_assert(TIsDerivedFrom<TEvent, FAuraEvent>::Value, "TEvent must derive from FAuraEvent");
        
        const FName EventName = TEvent::StaticStruct()->GetFName();
        
        if (EventHandlers.Contains(EventName))
        {
            for (auto& Handler : EventHandlers[EventName])
            {
                Handler(&Event);
            }
        }
        
        // 记录事件历史（用于调试）
        if (bEnableEventHistory)
        {
            RecordEvent(EventName, Event.Timestamp);
        }
    }
    
    // Blueprint支持
    UFUNCTION(BlueprintCallable, Category = "EventBus")
    void PublishGameplayTagEvent(FGameplayTag EventTag, AActor* Instigator);
    
    // 清除所有订阅
    UFUNCTION(BlueprintCallable, Category = "EventBus")
    void ClearAllSubscriptions();
    
    // 调试功能
    UFUNCTION(BlueprintCallable, Category = "EventBus|Debug")
    void EnableEventHistory(bool bEnable) { bEnableEventHistory = bEnable; }
    
    UFUNCTION(BlueprintCallable, Category = "EventBus|Debug")
    TArray<FString> GetEventHistory() const { return EventHistory; }
    
private:
    // 事件处理器映射
    TMap<FName, TArray<TFunction<void(const void*)>>> EventHandlers;
    
    // 事件历史（调试用）
    UPROPERTY()
    bool bEnableEventHistory = false;
    
    TArray<FString> EventHistory;
    
    void RecordEvent(FName EventName, float Timestamp);
};
```

#### 使用示例

```cpp
// AttributeSet中发布事件
void UAuraAttributeSet::PostGameplayEffectExecute(const FGameplayEffectModCallbackData& Data)
{
    Super::PostGameplayEffectExecute(Data);
    
    // 获取事件总线
    UEventBus* EventBus = GetWorld()->GetGameInstance()->GetSubsystem<UEventBus>();
    if (!EventBus) return;
    
    // 发布属性变化事件
    if (Data.EvaluatedData.Attribute == GetHealthAttribute())
    {
        FAttributeChangedEvent Event;
        Event.AttributeTag = FAuraGameplayTags::Get().Attributes_Vital_Health;
        Event.OldValue = GetHealth() - Data.EvaluatedData.Magnitude;
        Event.NewValue = GetHealth();
        Event.Instigator = Data.EffectSpec.GetEffectContext().GetInstigator();
        Event.Timestamp = GetWorld()->GetTimeSeconds();
        
        EventBus->Publish(Event);
    }
}

// WidgetController中订阅事件
void UOverlayWidgetController::BindCallbacksToDependencies()
{
    UEventBus* EventBus = GetWorld()->GetGameInstance()->GetSubsystem<UEventBus>();
    if (!EventBus) return;
    
    // 订阅属性变化事件
    EventBus->Subscribe<FAttributeChangedEvent>([this](const FAttributeChangedEvent& Event)
    {
        if (Event.AttributeTag == FAuraGameplayTags::Get().Attributes_Vital_Health)
        {
            OnHealthChanged.Broadcast(Event.NewValue);
        }
        else if (Event.AttributeTag == FAuraGameplayTags::Get().Attributes_Vital_Mana)
        {
            OnManaChanged.Broadcast(Event.NewValue);
        }
    });
    
    // 订阅伤害事件
    EventBus->Subscribe<FDamageEvent>([this](const FDamageEvent& Event)
    {
        // 显示伤害数字
        ShowDamageNumber(Event.Damage, Event.bCriticalHit, Event.Victim);
    });
}
```

### 方案2: 对象池系统（性能优化）

#### 问题分析
当前投射物、特效等对象频繁创建和销毁，导致：
- GC压力大
- 内存碎片化
- 运行时卡顿

#### 解决方案：通用对象池

```cpp
// ObjectPool.h - 新建文件
#pragma once

#include "CoreMinimal.h"
#include "Subsystems/WorldSubsystem.h"
#include "ObjectPool.generated.h"

/**
 * 可池化对象接口
 */
UINTERFACE(MinimalAPI, Blueprintable)
class UPoolableObject : public UInterface
{
    GENERATED_BODY()
};

class IPoolableObject
{
    GENERATED_BODY()
    
public:
    // 从池中获取时调用
    UFUNCTION(BlueprintNativeEvent, Category = "ObjectPool")
    void OnAcquiredFromPool();
    
    // 归还到池中时调用
    UFUNCTION(BlueprintNativeEvent, Category = "ObjectPool")
    void OnReturnedToPool();
    
    // 是否可以归还到池中
    UFUNCTION(BlueprintNativeEvent, Category = "ObjectPool")
    bool CanReturnToPool() const;
};

/**
 * 对象池配置
 */
USTRUCT(BlueprintType)
struct FObjectPoolConfig
{
    GENERATED_BODY()
    
    // 初始大小
    UPROPERTY(EditAnywhere, BlueprintReadWrite)
    int32 InitialSize = 10;
    
    // 最大大小（0表示无限制）
    UPROPERTY(EditAnywhere, BlueprintReadWrite)
    int32 MaxSize = 100;
    
    // 是否允许动态扩展
    UPROPERTY(EditAnywhere, BlueprintReadWrite)
    bool bAllowGrowth = true;
    
    // 自动回收时间（秒，0表示不自动回收）
    UPROPERTY(EditAnywhere, BlueprintReadWrite)
    float AutoReleaseTime = 0.f;
};

/**
 * 对象池管理器
 */
UCLASS()
class AURA_API UObjectPoolSubsystem : public UWorldSubsystem
{
    GENERATED_BODY()
    
public:
    virtual void Initialize(FSubsystemCollectionBase& Collection) override;
    virtual void Deinitialize() override;
    
    // 创建对象池
    UFUNCTION(BlueprintCallable, Category = "ObjectPool")
    void CreatePool(TSubclassOf<AActor> ActorClass, const FObjectPoolConfig& Config);
    
    // 从池中获取对象
    UFUNCTION(BlueprintCallable, Category = "ObjectPool")
    AActor* AcquireActor(TSubclassOf<AActor> ActorClass, const FTransform& SpawnTransform);
    
    // 归还对象到池
    UFUNCTION(BlueprintCallable, Category = "ObjectPool")
    void ReleaseActor(AActor* Actor);
    
    // 预热池（提前创建对象）
    UFUNCTION(BlueprintCallable, Category = "ObjectPool")
    void WarmupPool(TSubclassOf<AActor> ActorClass, int32 Count);
    
    // 清空池
    UFUNCTION(BlueprintCallable, Category = "ObjectPool")
    void ClearPool(TSubclassOf<AActor> ActorClass);
    
    // 获取池统计信息
    UFUNCTION(BlueprintCallable, Category = "ObjectPool|Debug")
    void GetPoolStats(TSubclassOf<AActor> ActorClass, int32& OutTotal, int32& OutActive, int32& OutInactive);
    
private:
    // 单个对象池
    struct FActorPool
    {
        TSubclassOf<AActor> ActorClass;
        FObjectPoolConfig Config;
        TArray<AActor*> InactiveActors;
        TArray<AActor*> ActiveActors;
        
        AActor* Acquire(UWorld* World, const FTransform& SpawnTransform);
        void Release(AActor* Actor);
        void Clear();
    };
    
    // 对象池映射
    TMap<TSubclassOf<AActor>, TSharedPtr<FActorPool>> Pools;
    
    // 获取或创建池
    TSharedPtr<FActorPool> GetOrCreatePool(TSubclassOf<AActor> ActorClass);
};
```

```cpp
// ObjectPool.cpp
#include "ObjectPool.h"

void UObjectPoolSubsystem::Initialize(FSubsystemCollectionBase& Collection)
{
    Super::Initialize(Collection);
    UE_LOG(LogTemp, Log, TEXT("ObjectPoolSubsystem Initialized"));
}

void UObjectPoolSubsystem::Deinitialize()
{
    // 清空所有池
    for (auto& Pair : Pools)
    {
        Pair.Value->Clear();
    }
    Pools.Empty();
    
    Super::Deinitialize();
}

void UObjectPoolSubsystem::CreatePool(TSubclassOf<AActor> ActorClass, const FObjectPoolConfig& Config)
{
    if (!ActorClass) return;
    
    TSharedPtr<FActorPool> Pool = MakeShared<FActorPool>();
    Pool->ActorClass = ActorClass;
    Pool->Config = Config;
    
    // 预创建对象
    UWorld* World = GetWorld();
    for (int32 i = 0; i < Config.InitialSize; ++i)
    {
        AActor* Actor = World->SpawnActor<AActor>(ActorClass);
        if (Actor)
        {
            Actor->SetActorHiddenInGame(true);
            Actor->SetActorEnableCollision(false);
            Actor->SetActorTickEnabled(false);
            Pool->InactiveActors.Add(Actor);
        }
    }
    
    Pools.Add(ActorClass, Pool);
    
    UE_LOG(LogTemp, Log, TEXT("Created pool for %s with %d objects"), 
           *ActorClass->GetName(), Config.InitialSize);
}

AActor* UObjectPoolSubsystem::AcquireActor(TSubclassOf<AActor> ActorClass, const FTransform& SpawnTransform)
{
    TSharedPtr<FActorPool> Pool = GetOrCreatePool(ActorClass);
    if (!Pool.IsValid()) return nullptr;
    
    return Pool->Acquire(GetWorld(), SpawnTransform);
}

void UObjectPoolSubsystem::ReleaseActor(AActor* Actor)
{
    if (!Actor) return;
    
    TSubclassOf<AActor> ActorClass = Actor->GetClass();
    TSharedPtr<FActorPool> Pool = GetOrCreatePool(ActorClass);
    if (Pool.IsValid())
    {
        Pool->Release(Actor);
    }
}

AActor* UObjectPoolSubsystem::FActorPool::Acquire(UWorld* World, const FTransform& SpawnTransform)
{
    AActor* Actor = nullptr;
    
    // 从非活动池中获取
    if (InactiveActors.Num() > 0)
    {
        Actor = InactiveActors.Pop();
    }
    // 如果允许扩展且未达到最大值，创建新对象
    else if (Config.bAllowGrowth && (Config.MaxSize == 0 || ActiveActors.Num() < Config.MaxSize))
    {
        Actor = World->SpawnActor<AActor>(ActorClass);
        UE_LOG(LogTemp, Warning, TEXT("Pool expanded for %s"), *ActorClass->GetName());
    }
    
    if (Actor)
    {
        // 重置Actor状态
        Actor->SetActorTransform(SpawnTransform);
        Actor->SetActorHiddenInGame(false);
        Actor->SetActorEnableCollision(true);
        Actor->SetActorTickEnabled(true);
        
        // 调用接口方法
        if (Actor->Implements<UPoolableObject>())
        {
            IPoolableObject::Execute_OnAcquiredFromPool(Actor);
        }
        
        ActiveActors.Add(Actor);
    }
    
    return Actor;
}

void UObjectPoolSubsystem::FActorPool::Release(AActor* Actor)
{
    if (!Actor) return;
    
    // 检查是否可以归还
    if (Actor->Implements<UPoolableObject>())
    {
        if (!IPoolableObject::Execute_CanReturnToPool(Actor))
        {
            Actor->Destroy();
            ActiveActors.Remove(Actor);
            return;
        }
    }
    
    // 重置Actor状态
    Actor->SetActorHiddenInGame(true);
    Actor->SetActorEnableCollision(false);
    Actor->SetActorTickEnabled(false);
    
    // 调用接口方法
    if (Actor->Implements<UPoolableObject>())
    {
        IPoolableObject::Execute_OnReturnedToPool(Actor);
    }
    
    // 移动到非活动池
    ActiveActors.Remove(Actor);
    InactiveActors.Add(Actor);
}
```

#### 投射物适配示例

```cpp
// AuraProjectile.h
UCLASS()
class AURA_API AAuraProjectile : public AActor, public IPoolableObject
{
    GENERATED_BODY()
    
public:
    // IPoolableObject接口实现
    virtual void OnAcquiredFromPool_Implementation() override;
    virtual void OnReturnedToPool_Implementation() override;
    virtual bool CanReturnToPool_Implementation() const override;
    
    // 设置投射物参数
    void InitializeProjectile(const FDamageEffectParams& Params, const FVector& Direction);
    
private:
    FTimerHandle LifetimeTimer;
    
    UFUNCTION()
    void OnLifetimeExpired();
};

// AuraProjectile.cpp
void AAuraProjectile::OnAcquiredFromPool_Implementation()
{
    // 重置投射物状态
    if (ProjectileMovement)
    {
        ProjectileMovement->Activate();
    }
}

void AAuraProjectile::OnReturnedToPool_Implementation()
{
    // 清理状态
    if (ProjectileMovement)
    {
        ProjectileMovement->Deactivate();
        ProjectileMovement->Velocity = FVector::ZeroVector;
    }
    
    // 清除定时器
    GetWorld()->GetTimerManager().ClearTimer(LifetimeTimer);
}

bool AAuraProjectile::CanReturnToPool_Implementation() const
{
    // 检查是否可以安全归还
    return !IsPendingKill() && GetWorld() != nullptr;
}

void AAuraProjectile::InitializeProjectile(const FDamageEffectParams& Params, const FVector& Direction)
{
    DamageEffectParams = Params;
    
    if (ProjectileMovement)
    {
        ProjectileMovement->Velocity = Direction * ProjectileMovement->InitialSpeed;
    }
    
    // 设置生命周期
    GetWorld()->GetTimerManager().SetTimer(
        LifetimeTimer,
        this,
        &AAuraProjectile::OnLifetimeExpired,
        Lifespan,
        false
    );
}

void AAuraProjectile::OnLifetimeExpired()
{
    // 归还到对象池
    UObjectPoolSubsystem* PoolSubsystem = GetWorld()->GetSubsystem<UObjectPoolSubsystem>();
    if (PoolSubsystem)
    {
        PoolSubsystem->ReleaseActor(this);
    }
}
```

```cpp
// AuraProjectileSpell.cpp
void UAuraProjectileSpell::SpawnProjectile(const FVector& ProjectileTargetLocation, const FGameplayTag& SocketTag)
{
    // 获取对象池
    UObjectPoolSubsystem* PoolSubsystem = GetWorld()->GetSubsystem<UObjectPoolSubsystem>();
    if (!PoolSubsystem) return;
    
    // 计算生成位置和方向
    const FVector SocketLocation = ICombatInterface::Execute_GetCombatSocketLocation(
        GetAvatarActorFromActorInfo(),
        SocketTag
    );
    
    FRotator Rotation = (ProjectileTargetLocation - SocketLocation).Rotation();
    
    FTransform SpawnTransform;
    SpawnTransform.SetLocation(SocketLocation);
    SpawnTransform.SetRotation(Rotation.Quaternion());
    
    // 从对象池获取投射物
    AAuraProjectile* Projectile = Cast<AAuraProjectile>(
        PoolSubsystem->AcquireActor(ProjectileClass, SpawnTransform)
    );
    
    if (Projectile)
    {
        // 初始化投射物
        Projectile->InitializeProjectile(
            MakeDamageEffectParamsFromClassDefaults(),
            Rotation.Vector()
        );
    }
}
```

```cpp
    }

    if (AbilityConfig->CastSound.IsValid())

void UConfigurableAbility::ExecuteAction(const FAbilityActionConfig& ActionConfig)
{
    // 如果有延迟，设置定时器
    if (ActionConfig.Delay > 0.f)
    {
        GetWorld()->GetTimerManager().SetTimer(
            ActionTimerHandle,
            [this, ActionConfig]()
            {
                ExecuteActionImmediate(ActionConfig);
            },
            ActionConfig.Delay,
            false
        );
    }
    else
    {
        ExecuteActionImmediate(ActionConfig);
    }
}

void UConfigurableAbility::ExecuteActionImmediate(const FAbilityActionConfig& ActionConfig)
{
    // 根据动作类型执行
    switch (ActionConfig.ActionType)
    {
    case EAbilityActionType::SpawnProjectile:
        ExecuteSpawnProjectile(ActionConfig);
        break;
        
    case EAbilityActionType::ApplyEffect:
        ExecuteApplyEffect(ActionConfig);
        break;
        
    case EAbilityActionType::PlayMontage:
        ExecutePlayMontage(ActionConfig);
        break;
        
    case EAbilityActionType::SpawnBeam:
        ExecuteSpawnBeam(ActionConfig);
        break;
        
    default:
        break;
    }
    
    // 执行下一个动作或结束技能
    CurrentActionIndex++;
    if (CurrentActionIndex < AbilityConfig->Actions.Num())
    {
        ExecuteAction(AbilityConfig->Actions[CurrentActionIndex]);
    }
    else
    {
        EndAbility(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo, true, false);
    }
}

void UConfigurableAbility::ExecuteSpawnProjectile(const FAbilityActionConfig& ActionConfig)
{
    // 获取对象池
    UObjectPoolSubsystem* PoolSubsystem = GetWorld()->GetSubsystem<UObjectPoolSubsystem>();
    if (!PoolSubsystem) return;
    
    // 获取生成位置
    const FVector SocketLocation = ICombatInterface::Execute_GetCombatSocketLocation(
        GetAvatarActorFromActorInfo(),
        ActionConfig.SocketTag
    );
    
    // 获取目标位置
    FVector TargetLocation = GetCursorTargetLocation();
    
    // 生成多个投射物
    for (int32 i = 0; i < ActionConfig.NumProjectiles; ++i)
    {
        // 计算扩散角度
        float Spread = ActionConfig.SpreadAngle;
        float AngleOffset = 0.f;
        
        if (ActionConfig.NumProjectiles > 1)
        {
            AngleOffset = FMath::Lerp(-Spread / 2.f, Spread / 2.f, 
                                      (float)i / (ActionConfig.NumProjectiles - 1));
        }
        
        // 计算方向
        FVector Direction = (TargetLocation - SocketLocation).GetSafeNormal();
        Direction = Direction.RotateAngleAxis(AngleOffset, FVector::UpVector);
        
        // 生成投射物
        FTransform SpawnTransform;
        SpawnTransform.SetLocation(SocketLocation);
        SpawnTransform.SetRotation(Direction.ToOrientationQuat());
        
        AActor* Projectile = PoolSubsystem->AcquireActor(
            ActionConfig.ProjectileClass,
            SpawnTransform
        );
        
        // 初始化投射物
        if (Projectile && Projectile->Implements<UProjectileInterface>())
        {
            IProjectileInterface::Execute_InitializeProjectile(
                Projectile,
                MakeDamageEffectParamsFromConfig(),
                Direction
            );
        }
    }
}
```
